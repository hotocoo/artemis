Examples

Basic Scan

# Scan current directory
artemis scan .

# Scan with high severity filter
artemis scan . --severity high

# Scan specific project
artemis scan /path/to/project --report json -o findings.json

CI/CD Integration

GitHub Actions

name: Security Scan
on: [push, pull_request]
jobs:
  scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '10.0.x'
      - run: dotnet tool install -g artemis
      - run: artemis scan . --report sarif -o findings.sarif
      - uses: github/codeql-action/upload-sarif@v3
        with:
          sarif_file: findings.sarif

GitLab CI

security-scan:
  stage: test
  script:
    - dotnet tool install -g artemis
    - artemis scan . --report sarif -o findings.sarif
  artifacts:
    reports:
      sarif: findings.sarif

Jenkins

pipeline {
    agent any
    stages {
        stage('Security Scan') {
            steps {
                sh 'dotnet tool install -g artemis'
                sh 'artemis scan . --report sarif -o findings.sarif'
            }
        }
    }
}

Custom Configuration

{
  "scanners": {
    "secrets": {
      "enabled": true,
      "patterns": [
        { "name": "AWS Key", "regex": "AKIA[0-9A-Z]{16}" }
      ]
    },
    "dependencies": {
      "enabled": true,
      "ecosystems": ["nuget", "npm"]
    }
  },
  "severity": "medium"
}

artemis audit --verify

# Export audit log
artemis audit --export audit-log.json
